-- How old is someone by surname (prefix match — Hill matches Hill, not Chhill)
SELECT TOP 5 CustomerId, Surname, Age
FROM Customers
WHERE Surname LIKE 'Hargrave%';

-- Where is someone from (Geography)
SELECT TOP 1 CustomerId, Surname, Geography
FROM Customers
WHERE Surname LIKE 'Hargrave%';

-- Credit score by surname
SELECT TOP 5 CustomerId, Surname, CreditScore
FROM Customers
WHERE Surname LIKE 'Hill%';

-- Lookup by customer id
SELECT TOP 1 CustomerId, Surname, Age, CreditScore
FROM Customers
WHERE CustomerId = 123456;

-- Average credit score by geography
SELECT Geography, AVG(CreditScore) AS avg_credit_score
FROM Customers
GROUP BY Geography;

-- Compare credit scores between two customers
SELECT TOP 10 CustomerId, Surname, CreditScore
FROM Customers
WHERE Surname LIKE 'Hargrave%' OR Surname LIKE 'Hill%'
ORDER BY CreditScore DESC;

-- Compare ages between two customers
SELECT TOP 10 CustomerId, Surname, Age
FROM Customers
WHERE Surname LIKE 'Hargrave%' OR Surname LIKE 'Hill%'
ORDER BY Age DESC;

-- Who has the higher balance
SELECT TOP 10 CustomerId, Surname, Balance
FROM Customers
WHERE Surname LIKE 'Hargrave%' OR Surname LIKE 'Hill%'
ORDER BY Balance DESC;

-- Compound "and" question: total customers AND male count (answer both parts)
SELECT
  COUNT(*) AS total_customers,
  SUM(CASE WHEN Gender = 'Male' THEN 1 ELSE 0 END) AS male_count
FROM Customers;

-- Compound "and" question: total customers AND average age
SELECT
  COUNT(*) AS total_customers,
  AVG(Age) AS average_age
FROM Customers;

-- Average salary AND five people closest to it (one query, no semicolon)
SELECT TOP 5
  CustomerId,
  Surname,
  EstimatedSalary,
  (SELECT AVG(EstimatedSalary) FROM Customers) AS average_salary
FROM Customers
ORDER BY ABS(EstimatedSalary - (SELECT AVG(EstimatedSalary) FROM Customers));

-- One person, multiple attributes: age AND credit score
SELECT TOP 5 CustomerId, Surname, Age, CreditScore
FROM Customers
WHERE Surname LIKE 'Hargrave%';

-- Three stats in one question: males, females, and average age
SELECT
  SUM(CASE WHEN Gender = 'Male' THEN 1 ELSE 0 END) AS male_count,
  SUM(CASE WHEN Gender = 'Female' THEN 1 ELSE 0 END) AS female_count,
  AVG(Age) AS average_age
FROM Customers;
